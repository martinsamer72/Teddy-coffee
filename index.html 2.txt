// LOCAL STORAGE — profiles & extra/deleted students
// ═══════════════════════════════════════════════════════════════════
function loadProfiles()  { try { return JSON.parse(localStorage.getItem('student_profiles')||'{}'); } catch { return {}; } }
function saveProfiles(p) { localStorage.setItem('student_profiles', JSON.stringify(p)); }

function loadExtra()   { try { return JSON.parse(localStorage.getItem('extra_students')||'{}'); }   catch { return {}; } }
function saveExtra(e)  { localStorage.setItem('extra_students', JSON.stringify(e)); }

function loadDeleted()   { try { return JSON.parse(localStorage.getItem('deleted_students')||'[]'); } catch { return []; } }
function saveDeleted(d)  { localStorage.setItem('deleted_students', JSON.stringify(d)); }

// ═══════════════════════════════════════════════════════════════════
// APPLY STORED EXTRA / DELETED STUDENTS ON BOOT
// ═══════════════════════════════════════════════════════════════════
function applyStoredStudents() {
  const extra   = loadExtra();
  const deleted = loadDeleted();
  CLASSES.forEach(cls => {
    deleted.forEach(key => {
      const [dcid, dname] = key.split('::');
      if (dcid === cls.id) {
        const idx = cls.students.indexOf(dname);
        if (idx !== -1) cls.students.splice(idx, 1);
      }
    });
    (extra[cls.id] || []).forEach(name => {
      if (!cls.students.includes(name)) cls.students.push(name);
    });
  });
}
applyStoredStudents();

// ═══════════════════════════════════════════════════════════════════
// PENDING QUEUE — retry failed profile saves
// ═══════════════════════════════════════════════════════════════════
function loadPending()  { try { return JSON.parse(localStorage.getItem('pending_profiles')||'[]'); } catch { return []; } }
function savePending(q) { localStorage.setItem('pending_profiles', JSON.stringify(q)); }

function queuePending(studentKey, profileData) {
  const q = loadPending();
  const idx = q.findIndex(x => x.key === studentKey);
  const entry = { key: studentKey, profile: profileData, ts: Date.now() };
  if (idx !== -1) q[idx] = entry; else q.push(entry);
  savePending(q);
}

async function flushPending() {
  const q = loadPending();
  if (!q.length) return;
  const remaining = [];
  for (const item of q) {
    try {
      await sbFetch('student_profiles', {
        method: 'POST',
        prefer: 'resolution=merge-duplicates,return=minimal',
        headers: { 'Prefer': 'resolution=merge-duplicates,return=minimal' },
        body: { student_key: item.key, profile: item.profile, updated_at: new Date().toISOString() }
      });
    } catch(e) { remaining.push(item); }
  }
  savePending(remaining);
}

// ═══════════════════════════════════════════════════════════════════
// PROFILE CRUD
// ═══════════════════════════════════════════════════════════════════
function profileKey(cid, name) { return cid + '::' + name; }
function getProfile(cid, name) {
  const p = loadProfiles();
  return p[profileKey(cid, name)] || {};
}

async function setProfile(cid, name, data) {
  // 1. Always save locally first (never lose data)
  const p = loadProfiles();
  const key = profileKey(cid, name);
  p[key] = data;
  saveProfiles(p);

  // 2. Queue in case offline
  queuePending(key, data);

  // 3. Try cloud push
  try {
    await sbFetch('student_profiles', {
      method: 'POST',
      prefer: 'resolution=merge-duplicates,return=minimal',
      headers: { 'Prefer': 'resolution=merge-duplicates,return=minimal' },
      body: { student_key: key, profile: data, updated_at: new Date().toISOString() }
    });
    // Remove from pending since push succeeded
    const q = loadPending();
    const filtered = q.filter(x => x.key !== key);
    savePending(filtered);
    return true;
  } catch(e) {
    console.warn('Profile push failed, queued for retry:', e);
    return false;
  }
}

// ═══════════════════════════════════════════════════════════════════
// CLOUD PULL
// ═══════════════════════════════════════════════════════════════════
async function pullProfilesFromCloud() {
  try {
    const rows = await sbFetch('student_profiles?select=*');
    if (!rows || !rows.length) return;
    const local = loadProfiles();
    rows.forEach(row => {
      // Cloud wins (latest updated_at), but never wipe a key that isn't in cloud
      local[row.student_key] = row.profile;
    });
    saveProfiles(local);
  } catch(e) { console.warn('Profile pull failed:', e); }
}

async function pullExtraStudentsFromCloud() {
  try {
    const rows = await sbFetch('extra_students?select=*');
    if (!rows || !rows.length) return;
    const extra   = loadExtra();
    const deleted = loadDeleted();
    rows.forEach(row => {
      const cls = CLASSES.find(c => c.id === row.class_id);
      if (!cls) return;
      const key = row.class_id + '::' + row.name;
      if (deleted.includes(key)) return;
      if (!cls.students.includes(row.name)) cls.students.push(row.name);
      if (!extra[row.class_id]) extra[row.class_id] = [];
      if (!extra[row.class_id].includes(row.name)) extra[row.class_id].push(row.name);
    });
    saveExtra(extra);
  } catch(e) { /* table may not exist */ }
}

async function pushExtraStudent(cid, name) {
  try {
    await sbFetch('extra_students', {
      method: 'POST',
      prefer: 'resolution=merge-duplicates,return=minimal',
      headers: { 'Prefer': 'resolution=merge-duplicates,return=minimal' },
      body: { class_id: cid, name: name }
    });
  } catch(e) { console.warn('Extra student cloud push failed:', e); }
}

function loadAttData() { try { return JSON.parse(localStorage.getItem('att_data')||'{}'); } catch { return {}; } }
function getAllAbsences(cid, name) {
  const rec = loadAttData()[cid] || {};
  return Object.values(rec).filter(day => day[name] === 'absent').length;
}

// SYNC STATUS
// ═══════════════════════════════════════════════════════════════════
let syncStatus = 'idle';
function setSyncStatus(s) {
  syncStatus = s;
  const el = document.getElementById('sync-indicator');
  const retry = document.getElementById('retry-btn');
  if (!el) return;
  const states = {
    syncing: { dot: '#f59e0b', text: 'جارٍ المزامنة…', anim: true },
    ok:      { dot: '#22c55e', text: 'تمت المزامنة ✓', anim: false },
    error:   { dot: '#ef4444', text: 'غير متصل',       anim: false },
    idle:    { dot: '#64748b', text: 'المزامنة السحابية', anim: false }
  };
  const st = states[s] || states.idle;
  const dotStyle = st.anim
    ? 'width:7px;height:7px;border-radius:50%;background:' + st.dot + ';display:inline-block;animation:pulse-dot 1.5s infinite'
    : 'width:7px;height:7px;border-radius:50%;background:' + st.dot + ';display:inline-block';
  el.innerHTML = '<span style="' + dotStyle + '"></span>' + st.text;
  if (retry) retry.style.display = (s === 'error') ? 'inline' : 'none';

  // Show pending count if any
  const pending = loadPending();
  if (pending.length && s !== 'syncing') {
    el.innerHTML += ' <span class="pending-dot" title="' + pending.length + ' تغييرات غير مرسلة" style="margin-right:4px"></span>';
  }
}

async function manualSync() {
  setSyncStatus('syncing');
  try {
    await flushPending();
    await pullProfilesFromCloud();
    await pullExtraStudentsFromCloud();
    setSyncStatus('ok');
    renderGrid();
    renderBirthdays();
    setTimeout(() => setSyncStatus('idle'), 3000);
  } catch(e) {
    setSyncStatus('error');
  }
}

function renderGrid() {
  const grid = document.getElementById('profiles-grid');
  const label = document.getElementById('grid-label');
  const countLbl = document.getElementById('count-label');
  if (!grid) return;

  const query = (document.getElementById('search-input').value || '').trim().toLowerCase();
  const classes = currentFilter === 'all' ? CLASSES : CLASSES.filter(c => c.id === currentFilter);

  grid.innerHTML = '';
  let count = 0;

  classes.forEach(cls => {
    cls.students.forEach((name, i) => {
      if (query && !name.toLowerCase().includes(query)) return;
      const absences = getAllAbsences(cls.id, name);
      const color = AV_COLORS[(cls.students.indexOf(name)) % AV_COLORS.length];
      const absColor = absences >= 2 ? 'var(--red)' : absences === 1 ? 'var(--amber)' : 'var(--green)';
      const absText = absences > 0 ? absences + ' غياب' : '✓ حسناً';

      const card = document.createElement('div');
      card.className = 'profile-card';
      card.style.animationDelay = (count * 0.04) + 's';
      card.onclick = () => openDetail(cls.id, name);
      card.innerHTML =
        '<div class="pc-avatar" style="background:' + color + '">' + initials(name) + '</div>' +
        '<div class="pc-name">' + name + '</div>' +
        '<div class="pc-class">' + cls.icon + ' ' + cls.sub + '</div>' +
        '<div class="pc-abs" style="color:' + absColor + '">' + absText + '</div>';
      grid.appendChild(card);
      count++;
    });
  });

  if (count === 0) {
    grid.innerHTML = '<div class="empty-state"><div style="font-size:36px;margin-bottom:10px">🔍</div>لا توجد نتائج</div>';
  }

  const totalStudents = CLASSES.reduce((a, c) => a + c.students.length, 0);
  countLbl.textContent = totalStudents + ' مخدوم · ' + CLASSES.length + ' فصول';
  label.textContent = 'المخدومين (' + count + ')';
}

// BIRTHDAYS THIS MONTH
// ═══════════════════════════════════════════════════════════════════
function renderBirthdays() {
  const container = document.getElementById('birthdays-section');
  if (!container) return;

  const now = new Date();
  const currentMonth = now.getMonth();
  const currentDay = now.getDate();
  const MONTHS_SHORT = ['jan','feb','mar','apr','may','jun','jul','aug','sep','oct','nov','dec'];
  const MONTHS_FULL  = ['january','february','march','april','may','june','july','august','september','october','november','december'];

  function parseDob(dob) {
    if (!dob) return null;
    dob = dob.trim().toLowerCase();
    let m = dob.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (m) return { month: parseInt(m[2])-1, day: parseInt(m[3]) };
    m = dob.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})$/);
    if (m) return { month: parseInt(m[2])-1, day: parseInt(m[1]) };
    for (let i=0; i<12; i++) {
      if (dob.includes(MONTHS_SHORT[i]) || dob.includes(MONTHS_FULL[i])) {
        const nums = dob.match(/\d+/g) || [];
        const day = nums.find(n => parseInt(n)<=31 && parseInt(n)>=1) || 1;
        return { month: i, day: parseInt(day) };
      }
    }
    return null;
  }

  const bdays = [];
  CLASSES.forEach(cls => {
    cls.students.forEach((name, i) => {
      const p = getProfile(cls.id, name);
      const parsed = parseDob(p.dob);
      if (parsed && parsed.month === currentMonth) {
        bdays.push({
          name, cid: cls.id, color: AV_COLORS[i % AV_COLORS.length],
          day: parsed.day, isToday: parsed.day === currentDay, classIcon: cls.icon
        });
      }
    });
  });

  if (!bdays.length) { container.innerHTML = ''; return; }
  bdays.sort((a, b) => a.day - b.day);

  const monthName = now.toLocaleString('ar-EG', { month: 'long' });
  let html = '<div class="birthday-section">' +
    '<div class="birthday-header"><span>🎉</span><span>' + bdays.length + ' مخدوم لديهم أعياد ميلاد في ' + monthName + '</span></div>';

  bdays.forEach(b => {
    const todayBadge = b.isToday
      ? '<span style="background:var(--gold);color:#1a1a1a;font-size:10px;font-weight:800;padding:2px 8px;border-radius:10px;margin-right:6px">🎂 اليوم!</span>'
      : '';
    const shadowStyle = b.isToday ? ';box-shadow:0 0 0 3px var(--gold),0 0 12px rgba(245,200,66,0.4)' : '';
    html +=
      '<div class="birthday-row" onclick="openDetail(\'' + b.cid + '\',\'' + b.name.replace(/'/g,"\\'") + '\')">' +
        '<div style="width:40px;height:40px;border-radius:50%;background:' + b.color + ';display:flex;align-items:center;justify-content:center;font-size:13px;font-weight:800;color:white;flex-shrink:0' + shadowStyle + '">' + initials(b.name) + '</div>' +
        '<div style="flex:1">' +
          '<div style="font-size:14px;font-weight:700;color:var(--white);display:flex;align-items:center;flex-wrap:wrap;gap:4px">' + b.name + todayBadge + '</div>' +
          '<div style="font-size:11px;color:var(--muted);margin-top:2px">' + b.classIcon + ' · ' + monthName + ' ' + b.day + '</div>' +
        '</div>' +
        '<span style="font-size:20px">' + (b.isToday ? '🎉' : '🎂') + '</span>' +
      '</div>';
  });
  html += '</div><div style="height:6px"></div>';
  container.innerHTML = html;
}

async function confirmAddPerson() {
  const nameEl  = document.getElementById('add-name');
  const classEl = document.getElementById('add-class');
  if (!nameEl || !classEl) return;

  const name = nameEl.value.trim();
  const cid  = classEl.value;
  if (!name) { nameEl.style.borderColor = 'var(--red)'; nameEl.focus(); return; }

  const cls = CLASSES.find(c => c.id === cid);
  if (cls.students.includes(name)) {
    closeModal();
    showModal(
      '<div class="modal-icon">⚠️</div>' +
      '<div class="modal-title">موجود مسبقاً</div>' +
      '<div class="modal-body">' + name + ' موجود بالفعل في ' + cls.name + '</div>' +
      '<button class="modal-btn primary" id="dup-ok">حسناً</button>'
    );
    document.getElementById('dup-ok').addEventListener('click', closeModal);
    return;
  }

  // Add locally
  cls.students.push(name);
  const extra = loadExtra();
  if (!extra[cid]) extra[cid] = [];
  if (!extra[cid].includes(name)) extra[cid].push(name);
  saveExtra(extra);

  // Remove from deleted if re-adding
  const deleted = loadDeleted();
  const key = cid + '::' + name;
  const di = deleted.indexOf(key);
  if (di !== -1) { deleted.splice(di, 1); saveDeleted(deleted); }

  // Push to cloud
  pushExtraStudent(cid, name);

  closeModal();
  showModal(
    '<div class="modal-icon">✅</div>' +
    '<div class="modal-title">تمت الإضافة</div>' +
    '<div class="modal-body"><strong style="color:white">' + name + '</strong> تمت إضافته إلى <strong style="color:white">' + cls.name + '</strong></div>' +
    '<button class="modal-btn primary" id="added-ok">تم</button>'
  );
  document.getElementById('added-ok').addEventListener('click', function() {
    closeModal();
    renderGrid();
    renderBirthdays();
  });
}

  );
  document.getElementById('del-confirm').addEventListener('click', () => doDelete(cid, name));
  document.getElementById('del-cancel').addEventListener('click', closeModal);
}

async function doDelete(cid, name) {
  const cls = CLASSES.find(c => c.id === cid);

  // Remove from memory
  const idx = cls.students.indexOf(name);
  if (idx !== -1) cls.students.splice(idx, 1);

  // Remove from extra
  const extra = loadExtra();
  if (extra[cid]) {
    const ei = extra[cid].indexOf(name);
    if (ei !== -1) extra[cid].splice(ei, 1);
    saveExtra(extra);
  }

  // Add to deleted
  const deleted = loadDeleted();
  const key = cid + '::' + name;
  if (!deleted.includes(key)) { deleted.push(key); saveDeleted(deleted); }

  // Cloud delete
  try {
    await sbFetch('extra_students?class_id=eq.' + encodeURIComponent(cid) + '&name=eq.' + encodeURIComponent(name), {
      method: 'DELETE', headers: { 'Prefer': 'return=minimal' }
    });
  } catch(e) { console.warn('Cloud delete failed:', e); }

  closeModal();
  showModal(
    '<div class="modal-icon">✅</div>' +
    '<div class="modal-title">تم الحذف</div>' +
    '<div class="modal-body"><strong style="color:white">' + name + '</strong> تمت إزالته من ' + cls.name + '</div>' +
    '<button class="modal-btn primary" id="del-done">تم</button>'
  );
  document.getElementById('del-done').addEventListener('click', function() {
    closeModal();
    showScreen('screen-list');
    renderGrid();
    renderBirthdays();
  });
}

// MODAL HELPERS
// ═══════════════════════════════════════════════════════════════════
function showModal(html) {
  const m = document.getElementById('modal');
  m.innerHTML = '<div class="modal-overlay" onclick="handleOverlayClick(event)"><div class="modal-box">' + html + '</div></div>';
  m.style.display = 'block';
}
function closeModal() {
  const m = document.getElementById('modal');
  m.style.display = 'none';
  m.innerHTML = '';
}
function handleOverlayClick(e) {
  if (e.target.classList.contains('modal-overlay')) closeModal();
}

// ═══════════════════════════════════════════════════════════════════
// PERIODIC SYNC — every 60 seconds when visible
// ═══════════════════════════════════════════════════════════════════
let syncInterval = null;
function startSyncPoll() {
  if (syncInterval) return;
  syncInterval = setInterval(async function() {
    if (document.visibilityState !== 'visible') return;
    await flushPending();
    await pullProfilesFromCloud();
    // Re-render grid if on list screen
    if (document.getElementById('screen-list').classList.contains('active')) {
      renderGrid();
      renderBirthdays();
    }
  }, 60000);
}

// Flush pending when coming back online
window.addEventListener('online', async function() {
  setSyncStatus('syncing');
  await flushPending();
  await pullProfilesFromCloud();
  renderGrid();
  setSyncStatus('ok');
  setTimeout(() => setSyncStatus('idle'), 3000);
});

// Warn when going offline
window.addEventListener('offline', function() {
  setSyncStatus('error');
});

// INIT

async function init() {
  renderGrid();
  renderBirthdays();
  setSyncStatus('syncing');

  try {
    // Flush any locally queued profiles first
    await flushPending();
    // Pull latest from cloud
    await pullProfilesFromCloud();
    await pullExtraStudentsFromCloud();
    // Re-render with fresh data
    renderGrid();
    renderBirthdays();
    setSyncStatus('ok');
    setTimeout(() => setSyncStatus('idle'), 3000);
  } catch(e) {
    setSyncStatus('error');
  }

  startSyncPoll();
}
